// Helpers de fecha local para evitar desfases por zona horaria
const getLocalDateStr = (date: Date) => {
  const year = date.getUTCFullYear();
  const month = String(date.getUTCMonth() + 1).padStart(2, "0");
  const day = String(date.getUTCDate()).padStart(2, "0");
  return `${year}-${month}-${day}`;
};
const toLocalDateInputValue = (isoString: string) => {
  return getLocalDateStr(new Date(isoString));
};
const todayLocalStr = () => getLocalDateStr(new Date());
 
export const TransactionDialog = ({ 
  open, 
  onOpenChange, 
  mode, 
  transaction, 
  onSave,
  currency 
}: TransactionDialogProps) => {
  const { categories, isLoading: isLoadingCategories } = useCategories();
  const { activePaymentMethods, isLoading: isLoadingPaymentMethods } = usePaymentMethods();

  const [categoria, setCategoria] = useState(transaction?.categoria || "");
  const [subcategoria, setSubcategoria] = useState(transaction?.subcategoria || "");

  const [establecimiento, setEstablecimiento] = useState(transaction?.establecimiento || "");
  const [medioPago, setMedioPago] = useState<string>(transaction?.medio_pago || "");
  const [moneda, setMoneda] = useState<string>(transaction?.moneda || currency || "COP");
  const [archivo, setArchivo] = useState<File | null>(null);
  const [archivoUrl, setArchivoUrl] = useState(transaction?.archivo_url || "");
  const [archivoNombre, setArchivoNombre] = useState(transaction?.archivo_nombre || "");
  const [uploading, setUploading] = useState(false);
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [additionalInfoOpen, setAdditionalInfoOpen] = useState(false);

  const initializedForTransactionId = useRef<string | null>(null);
  
  // Set default payment method when loaded

    if (activePaymentMethods.length > 0 && !medioPago && mode === "add") {
      setMedioPago(activePaymentMethods[0].nombre);
    }
  }, [activePaymentMethods, medioPago, mode]);
 
  useEffect(() => {
    // Para modo edición: solo inicializar si es una transacción diferente
    if (open && transaction && mode === "edit") {
      const transactionId = transaction.id;
      
      // Solo inicializar si es una transacción diferente a la que ya inicializamos
      if (initializedForTransactionId.current !== transactionId) {
        initializedForTransactionId.current = transactionId;
        
        setTipo(transaction.tipo);
        setCategoria(transaction.categoria);
        setSubcategoria(transaction.subcategoria || "");
        setValor(transaction.valor);
        const dateStr = transaction.fecha ? toLocalDateInputValue(transaction.fecha) : todayLocalStr();
        setFecha(dateStr);
        setDescripcion(transaction.descripcion || "");
        setEstablecimiento(transaction.establecimiento || "");
        setMedioPago(transaction.medio_pago);
        setMoneda(transaction.moneda || currency);
        setArchivoUrl(transaction.archivo_url || "");
        setArchivoNombre(transaction.archivo_nombre || "");
        // Abrir acordeón si hay información adicional
        if (transaction.establecimiento || transaction.descripcion || transaction.archivo_url) {
          setAdditionalInfoOpen(true);
        }
      }
    } else if (open && mode === "add") {
      // Reset para modo agregar
      initializedForTransactionId.current = null;
      resetForm();
    } else if (!open) {
      // Cuando el diálogo se cierra, limpiar el ref para permitir re-inicialización
      initializedForTransactionId.current = null;
    }
  }, [open, transaction?.id, mode]);
 
  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (file) {
      if (file.size > 5 * 1024 * 1024) {
        toast.error("El archivo no puede superar los 5MB");
        return;
      }
      setArchivo(file);
    }
  };
 
  const removeFile = () => {
    setArchivo(null);
    setArchivoUrl("");
    setArchivoNombre("");
  };
 
  const uploadFile = async (file: File): Promise<{ url: string; nombre: string } | null> => {
    try {
      await new Promise(resolve => setTimeout(resolve, 500));
      const url = URL.createObjectURL(file);
      return { url, nombre: file.name };
    } catch (error) {
      console.error('Error uploading file:', error);
      toast.error("Error al subir el archivo");
      return null;
    }
  };
 
  const resetForm = () => {
    setTipo("ingreso");
    setCategoria("");
    setSubcategoria("");
    setValor(undefined);
    setFecha(todayLocalStr());
    setDescripcion("");
    setEstablecimiento("");
    setMedioPago(activePaymentMethods[0]?.nombre || "");
    setMoneda(currency);
    setArchivo(null);
    setArchivoUrl("");
    setArchivoNombre("");
    setAdditionalInfoOpen(false);
  };
 
  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    
    if (isSubmitting) return;
    
    if (!categoria || !valor) {
      toast.error("Por favor completa todos los campos requeridos (Categoría, Valor)");
      return;
    }
    
    // Capturar todos los datos ANTES de cualquier operación async
    const fileUrl = archivoUrl;
    const fileName = archivoNombre;
 
    // Para transacciones normales
    const transactionData = {
      tipo,
      categoria,
      subcategoria: subcategoria || undefined,
      valor: valor || 0,
      moneda: moneda,
      fecha,
      descripcion: descripcion || `${tipo === "ingreso" ? "Ingreso" : "Gasto"} - ${categoria}`,
      establecimiento: establecimiento || undefined,
      medio_pago: medioPago,
      archivo_url: fileUrl || undefined,
      archivo_nombre: fileName || undefined
    };
 
    // Capturar datos para el toast ANTES de cerrar
    const savedValor = valor || 0;
    const savedDescripcion = descripcion || categoria;
    const savedTipo = tipo;
    const savedMoneda = moneda;
    const isEdit = mode === "edit" && transaction;
    const fullTransactionData = isEdit ? { ...transaction, ...transactionData } : transactionData;
 
    // Cerrar dialog INMEDIATAMENTE - esto es síncrono
    onOpenChange(false);
    
    if (!isEdit) {
      resetForm();
    }
 
    // Ejecutar la mutación DESPUÉS del cierre del dialog
    Promise.resolve(onSave(fullTransactionData))
      .then(() => {
        if (isEdit) {
          toast.success("Transacción actualizada exitosamente");
        } else {
          toast.success(
            savedTipo === "ingreso" 
              ? "¡Ingreso registrado exitosamente!" 
              : "¡Gasto registrado exitosamente!",
            {
              description: `${savedDescripcion} - ${new Intl.NumberFormat('es-CO', {
                style: 'currency',
                currency: savedMoneda,
                minimumFractionDigits: 0,
              }).format(savedValor)}`
            }
          );
        }
      })
      .catch((error) => {
        console.error("Error guardando transacción:", error);
        toast.error("Error al guardar transacción");
      });
  };
 
  const filteredCategories = categories
    .filter(cat => cat.tipo === tipo)
    .sort((a, b) => {
      // Colocar "Otros" al final
      if (a.nombre === "Otros") return 1;
      if (b.nombre === "Otros") return -1;
      return 0;
    });
  const selectedCategory = categories.find(cat => cat.nombre === categoria);
  const subcategorias: Subcategory[] = selectedCategory?.subcategorias || [];
 
  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent className="sm:max-w-[500px] max-h-[90vh] overflow-y-auto">
        <DialogHeader>
          <DialogTitle>{mode === "edit" ? "Editar transacción" : "Nueva transacción"}</DialogTitle>

                variant={tipo === "ingreso" ? "default" : "outline"}
                className={cn(
                  "h-16 flex flex-col gap-1",
                  tipo === "ingreso" && "bg-income hover:bg-income/90"
                )}

                  setTipo("ingreso");
                  setCategoria("");
                  setSubcategoria("");

              >
                <TrendingUp className="h-5 w-5" />
                <span className="text-xs">Ingreso</span>
              </Button>

                  setTipo("gasto");
                  setCategoria("");

              <div className="text-center p-8 text-muted-foreground">
                No hay categorías de {tipo} disponibles
              </div>
            ) : (
              <div className="grid grid-cols-2 sm:grid-cols-3 gap-2">
                {filteredCategories.map((cat) => {
                  const IconComponent = iconMap[cat.icon] || iconMap.MoreHorizontal;

                  return (
                    <Button

                          isSelected ? "bg-background/90" : "bg-muted/50"

                      >
                        <IconComponent 
                          className="h-7 w-7"

                            color: '#FF6B35',
                            filter: isSelected ? 'drop-shadow(0 0 8px #FF6B35)' : 'none'

                      </div>
                      <span className={cn(
                        "text-xs font-medium",

                    className={cn(
                      "h-auto py-3 text-sm",
                      subcategoria === sub.nombre && "ring-2 ring-ring"
                    )}
                    onClick={() => setSubcategoria(sub.nombre)}
                  >
                    {sub.nombre}
                  </Button>
                ))}
              </div>
            </div>
          )}

                  value={moneda} 
                  onValueChange={setMoneda}
                  compact

                <CurrencyInput
                  id="valor"

                  placeholder="0"

                />

              />

          <div className="space-y-2">

            <div className="grid grid-cols-2 sm:grid-cols-3 gap-2">
              {isLoadingPaymentMethods ? (
                <div className="col-span-full flex items-center justify-center p-4">
                  <div className="animate-spin rounded-full h-6 w-6 border-b-2 border-primary"></div>
                </div>
              ) : activePaymentMethods.length === 0 ? (
                <div className="col-span-full text-center p-4 text-muted-foreground">
                  No hay medios de pago activos
                </div>
              ) : (
                activePaymentMethods.map((method) => {
                  const IconComponent = iconMap[method.icono] || Wallet;
                  const isSelected = medioPago === method.nombre;
                  return (
                    <Button
                      key={method.id}
                      type="button"
                      variant={isSelected ? "default" : "outline"}
                      className={cn(
                        "h-24 flex flex-col gap-2 relative overflow-hidden transition-all",
                        isSelected && "ring-2 ring-primary shadow-lg scale-105"
                      )}
                      onClick={() => setMedioPago(method.nombre)}
                    >
                      <div 
                        className={cn(
                          "rounded-lg p-2 transition-all",
                          isSelected ? "bg-background/90" : "bg-muted/50"
                        )}
                      >
                        <IconComponent 
                          className="h-7 w-7"
                          style={{ 
                            color: method.color,
                            filter: isSelected ? `drop-shadow(0 0 8px ${method.color})` : 'none'
                          }} 
                        />
                      </div>
                      <span className={cn(
                        "text-xs font-medium line-clamp-1",

                        {method.nombre}

                    </Button>
                  );

              )}
            </div>
          </div>
 
          {/* Información adicional - Acordeón */}

                className="w-full flex items-center justify-between p-3 h-auto hover:bg-muted/50"
              >
                <span className="text-sm font-medium text-muted-foreground">
                  Información adicional (opcional)
                </span>
                <ChevronDown className={cn(
                  "h-4 w-4 text-muted-foreground transition-transform",
                  additionalInfoOpen && "transform rotate-180"
                )} />
              </Button>
            </CollapsibleTrigger>
            <CollapsibleContent className="space-y-4 pt-2">
              {/* Establecimiento */}
              <div className="space-y-2">
                <Label htmlFor="establecimiento" className="flex items-center gap-2">
                  <Store className="h-4 w-4 text-muted-foreground" />
                  Establecimiento
                </Label>
                <Input
                  id="establecimiento"
                  placeholder="Ej: Éxito, Rappi, Starbucks..."
                  value={establecimiento}
                  onChange={(e) => setEstablecimiento(e.target.value)}
                />
              </div>
 
              {/* Descripción */}
              <div className="space-y-2">
                <Label htmlFor="descripcion">Descripción</Label>
                <Textarea
                  id="descripcion"
                  placeholder="Ej: Almuerzo con amigos"
                  value={descripcion}
                  onChange={(e) => setDescripcion(e.target.value)}
                />
              </div>
 
              {/* Archivo adjunto */}
              <div className="space-y-2">
                <Label htmlFor="archivo">Adjuntar archivo</Label>
                <div className="flex items-center gap-2">
                  <Button
                    type="button"
                    variant="outline"
                    className="flex-1"
                    onClick={() => toast.info("Esta función estará activa próximamente")}
                  >
                    <Upload className="h-4 w-4 mr-2" />
                    Seleccionar archivo
                  </Button>
                </div>
                <p className="text-xs text-muted-foreground">
                  Esta función estará disponible próximamente
                </p>
              </div>
            </CollapsibleContent>
          </Collapsible>
 
          <div className="flex gap-3 pt-4">
            <Button type="button" variant="outline" onClick={() => onOpenChange(false)} className="flex-1" disabled={isSubmitting}>
              Cancelar
            </Button>
            <Button type="submit" className="flex-1" disabled={uploading || isSubmitting}>
              {isSubmitting ? "Guardando..." : uploading ? "Subiendo..." : mode === "edit" ? "Guardar cambios" : "Guardar"}
            </Button>
          </div>
        </form>
      </DialogContent>
    </Dialog>
  );
};
