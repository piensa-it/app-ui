import { Card } from "@/components/ui/card";
import { Progress } from "@/components/ui/progress";
import { Target } from "lucide-react";
import { formatCurrency } from "@/utils/formatCurrency";
 
interface GoalProgressProps {
  current: number;
  goal: number;
  currency: string;
  format?: "completo" | "miles";
}
 
export const GoalProgress = ({ current, goal, currency, format = "miles" }: GoalProgressProps) => {
  const percentage = goal > 0 ? Math.min((current / goal) * 100, 100) : 0;
  const remaining = Math.max(goal - current, 0);
 
  const getMessage = () => {
    if (goal === 0) return "🎯 Configura tu meta de ahorro mensual";
    if (percentage >= 100) return "🎉 ¡Meta alcanzada! ¡Excelente trabajo!";
    if (percentage >= 80) return "🔥 ¡Casi lo logras! Sigue así";
    if (percentage >= 50) return "💪 Vas por buen camino";
    if (percentage >= 25) return "📈 Buen comienzo, continúa";
    return "🎯 Comienza a ahorrar para tu meta";
  };
 
  return (
    <Card className="p-6 animate-fade-in">
      <div className="flex items-center gap-2 mb-4">
        <Target className="h-5 w-5 text-primary" />
        <h3 className="text-lg font-semibold">Meta de ahorro mensual</h3>
      </div>
      
      <div className="space-y-3">
        <div className="flex justify-between text-sm">
          <span className="text-muted-foreground">Ahorrado</span>
          <span className="font-semibold text-primary">{formatCurrency(current, currency, format)}</span>
        </div>
        
        <Progress value={percentage} className="h-3" />
        
        <div className="flex justify-between text-sm">
          <span className="text-muted-foreground">Meta</span>
          <span className="font-semibold">{formatCurrency(goal, currency, format)}</span>
        </div>
        
        <div className="pt-2 border-t border-border">
          <p className="text-sm font-medium text-center">{getMessage()}</p>
          {percentage < 100 && goal > 0 && (
            <p className="text-xs text-muted-foreground text-center mt-1">
              Te faltan {formatCurrency(remaining, currency, format)} para alcanzar tu meta
            </p>
          )}
        </div>
      </div>
    </Card>
  );
};
