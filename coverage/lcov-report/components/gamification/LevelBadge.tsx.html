import { cn } from "@/lib/utils";
import { Sprout, BookOpen, FolderCheck, Brain, Crown } from "lucide-react";
import { LEVEL_CONFIG } from "@/hooks/useGamification";
 
interface LevelBadgeProps {
  level: string;
  size?: "sm" | "md" | "lg";
  showLabel?: boolean;
  className?: string;
}
 
const LEVEL_ICONS: Record<string, React.ElementType> = {
  Sprout,
  BookOpen,
  FolderCheck,
  Brain,
  Crown,
};
 
export const LevelBadge = ({ level, size = "md", showLabel = true, className }: LevelBadgeProps) => {
  const config = LEVEL_CONFIG[level as keyof typeof LEVEL_CONFIG] || LEVEL_CONFIG.novato;
  const Icon = LEVEL_ICONS[config.icon] || Sprout;
 
  const sizeClasses = {
    sm: "h-6 w-6 text-xs",
    md: "h-8 w-8 text-sm",
    lg: "h-12 w-12 text-base",
  };
 
  const iconSizes = {
    sm: "h-3 w-3",
    md: "h-4 w-4",
    lg: "h-6 w-6",
  };
 
  return (
    <div className={cn("flex items-center gap-2", className)}>
      <div
        className={cn(
          "rounded-full flex items-center justify-center text-white",
          config.color,
          sizeClasses[size]
        )}
      >
        <Icon className={iconSizes[size]} />
      </div>
      {showLabel && (
        <span className={cn("font-medium", size === "sm" ? "text-xs" : size === "lg" ? "text-lg" : "text-sm")}>
          {config.name}
        </span>
      )}
    </div>
  );
};
